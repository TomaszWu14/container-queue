def test_po_dict_exposes_crd_target_and_deviation(client, admin_headers):

        co = db.query(Company).first()
        po = PurchaseOrder(company_id=co.id, order_no="PD-1",
                           crd=datetime.date(2026, 9, 25), crd_target=datetime.date(2026, 9, 10),
                           cart_status=CartStatus.w_koszyku.value)
        db.add(po); db.commit()
    r = client.get("/api/purchase-orders?cart_status=w_koszyku", headers=admin_headers)
    row = next(x for x in r.json() if x["order_no"] == "PD-1")
    assert row["crd_target"] == "2026-09-10"
    assert row["deviation_days"] == 15

```python
def _po_dict(po: PurchaseOrder) -> dict:
    from ..consolidation import crd_deviation_days
    return {"id": po.id, "order_no": po.order_no, "supplier": po.supplier,
            "cbm": float(po.cbm) if po.cbm is not None else None,
            "crd": po.crd, "crd_target": po.crd_target,
            "deviation_days": crd_deviation_days(po),
            "cart_status": po.cart_status, "container_id": po.container_id}
```
(jeśli `crd_deviation_days` już jest importowane u góry pliku — użyj tego importu zamiast lokalnego)
