```python
@router.get("/consolidation/containers")
def list_consolidation_containers(status: str = "otwarty",
                                  db: Session = Depends(get_db),
                                  user: User = Depends(get_current_user)) -> list[dict]:
    q = select(Container).where(Container.consolidation_status == status)
    ids = company_filter_ids(user)
    if ids is not None:
        q = q.where(Container.company_id.in_(ids))
    out = []
    for c in db.scalars(q.order_by(Container.container_no)).all():
        n = db.scalar(select(func.count(PurchaseOrder.id))
                      .where(PurchaseOrder.container_id == c.id))
        out.append({"id": c.id, "container_no": c.container_no,
                    "consolidation_status": c.consolidation_status,
                    "capacity_cbm": float(c.capacity_cbm),
                    "fill_cbm": container_fill_cbm(db, c.id),
                    "crd": container_crd(db, c.id), "order_count": int(n or 0)})
    return out
```
(dodaj `from sqlalchemy import func` jeśli brak; `container_fill_cbm/container_crd` już importowane)
